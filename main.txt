#include <iostream>
#include <vector>
#include <string>
#include <stdexcept>
#include "sqlite/sqlite3.h"
using namespace std;

// ====================== DATA STRUCTURES ======================
struct Course {
    string id;
    string name;
    int credits;
    bool hasLecture;
    bool hasLab;
    bool hasLongTut;
    bool hasShortTut;
};

struct Instructor {
    string id;
    string name;
    string role;
    string preferredSlots;
    string qualifiedCourses;
};

struct InstructorCourse {
    string instructorID;
    string courseID;
};

struct Room {
    string id;
    string building;
    string roomName;
    int capacity;
    string roomType;
};

struct TimeSlot {
    int id;
    string day;
    string slotType;
    string startTxt;
    string endTxt;
    int startMin;
    int endMin;
};

// ====================== DATABASE MANAGER ======================
class DatabaseManager {
private:
    sqlite3* db;

public:
    DatabaseManager(const string& dbFile) {
        if (sqlite3_open(dbFile.c_str(), &db) != SQLITE_OK) {
            throw runtime_error("Cannot open database: " + string(sqlite3_errmsg(db)));
        }
        cout << "✅ Connected to database successfully.\n";
    }

    ~DatabaseManager() {
        sqlite3_close(db);
        cout << "✅ Database connection closed.\n";
    }

    // ---------- Load Courses ----------
    vector<Course> getCourses() {
        vector<Course> list;
        const char* sql = "SELECT CourseID, CourseName, Credits, HasLecture, HasLab, HasLongTut, HasShortTut FROM Courses;";
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Course c;
            c.id = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
            c.name = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            c.credits = sqlite3_column_int(stmt, 2);
            c.hasLecture = sqlite3_column_int(stmt, 3);
            c.hasLab = sqlite3_column_int(stmt, 4);
            c.hasLongTut = sqlite3_column_int(stmt, 5);
            c.hasShortTut = sqlite3_column_int(stmt, 6);
            list.push_back(c);
        }
        sqlite3_finalize(stmt);
        return list;
    }

    // ---------- Load Instructors ----------
    vector<Instructor> getInstructors() {
        vector<Instructor> list;
        const char* sql = "SELECT InstructorID, Name, Role, PreferredSlots, QualifiedCourses FROM Instructor;";
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Instructor i;
            i.id = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
            i.name = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            i.role = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 2));
            i.preferredSlots = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 3));
            i.qualifiedCourses = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 4));
            list.push_back(i);
        }
        sqlite3_finalize(stmt);
        return list;
    }

    // ---------- Load InstructorCourses ----------
    vector<InstructorCourse> getInstructorCourses() {
        vector<InstructorCourse> list;
        const char* sql = "SELECT InstructorID, CourseID FROM InstructorCourses;";
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            InstructorCourse ic;
            ic.instructorID = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
            ic.courseID = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            list.push_back(ic);
        }
        sqlite3_finalize(stmt);
        return list;
    }

    // ---------- Load Rooms ----------
    vector<Room> getRooms() {
        vector<Room> list;
        const char* sql = "SELECT RoomID, Building, RoomName, Capacity, RoomType FROM Rooms;";
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            Room r;
            r.id = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 0));
            r.building = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            r.roomName = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 2));
            r.capacity = sqlite3_column_int(stmt, 3);
            r.roomType = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 4));
            list.push_back(r);
        }
        sqlite3_finalize(stmt);
        return list;
    }

    // ---------- Load TimeSlots ----------
    vector<TimeSlot> getTimeSlots() {
        vector<TimeSlot> list;
        const char* sql = "SELECT TimeSlotID, Day, SlotType, StartTimeTxt, EndTimeTxt, StartMin, EndMin FROM TimeSlots;";
        sqlite3_stmt* stmt;
        sqlite3_prepare_v2(db, sql, -1, &stmt, nullptr);

        while (sqlite3_step(stmt) == SQLITE_ROW) {
            TimeSlot t;
            t.id = sqlite3_column_int(stmt, 0);
            t.day = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 1));
            t.slotType = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 2));
            t.startTxt = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 3));
            t.endTxt = reinterpret_cast<const char*>(sqlite3_column_text(stmt, 4));
            t.startMin = sqlite3_column_int(stmt, 5);
            t.endMin = sqlite3_column_int(stmt, 6);
            list.push_back(t);
        }
        sqlite3_finalize(stmt);
        return list;
    }
};

int main() {
    try {
        DatabaseManager db("F:\\TestSql\\data\\database.db");


        vector<Course> courses = db.getCourses();
        vector<Instructor> instructors = db.getInstructors();
        vector<Room> rooms = db.getRooms();
        vector<TimeSlot> slots = db.getTimeSlots();

        cout << "\n=== Data Summary ===\n";
        cout << "Courses: " << courses.size() << endl;
        cout << "Instructors: " << instructors.size() << endl;
        cout << "Rooms: " << rooms.size() << endl;
        cout << "Time Slots: " << slots.size() << endl;

        cout << "\n--- Sample Course Data ---\n";
        for (auto& c : courses)
            cout << c.id << " | " << c.name << " | Credits: " << c.credits << endl;

        cout << "\n--- Sample Staff Data---\n";
        for (auto& staff : instructors)
            cout << staff.id << " | " << staff.name << " | " << staff.role << " | "
                << staff.preferredSlots << " | " << staff.qualifiedCourses << endl;

    }
    catch (const exception& e) {
        cerr << "❌ Error: " << e.what() << endl;
    }

    return 0;
}
